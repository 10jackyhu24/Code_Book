\usepackage{tcolorbox}
\tcbuselibrary{breakable} % Load breakable library to allow code blocks to span across pages
\usepackage{fvextra}      % Use fvextra (extension of fancyvrb) for advanced verbatim features including line breaking
\usepackage{xcolor}

% 1. Redefine Shaded environment: add background color and gray rounded border for code blocks (similar to md's ```)
\renewenvironment{Shaded}{
  \begin{tcolorbox}[
    colback=gray!6,           % Light gray background
    colframe=gray!30,          % Border color
    arc=3pt,                   % Corner radius
    boxrule=0.5pt,             % Border thickness
    left=5pt, right=5pt, top=5pt, bottom=5pt, % Padding / Inner margins
    boxsep=0pt,
    breakable                  % Allow page breaking
  ]
}{
  \end{tcolorbox}
}

% 2. Redefine Highlighting environment: set to monospaced font globally, show line numbers, and enable automatic line wrapping
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  commandchars=\\\{\},
  numbers=left,                % Show line numbers on the left
  numbersep=8pt,               % Distance between line numbers and code
  fontsize=\small,             % Font size
  fontfamily=tt,               % Use monospaced font
  breaklines=true,             % Automatically wrap long lines of code!
  breakanywhere=true           % Allow line breaks anywhere if a line is too long
}

% 3. Custom line number format: set to tiny and elegant gray
\renewcommand{\theFancyVerbLine}{\tiny\color{gray}\arabic{FancyVerbLine}}

\raggedbottom

% 4. Page numbering configuration: Move page number to the top right
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{} % Clear all header and footer fields

% Place page number on the top right
\rhead{\thepage} 

% Remove the default horizontal rule under the header
\renewcommand{\headrulewidth}{0pt}

% Force the first page to also use this style
\fancypagestyle{plain}{
  \fancyhf{}
  \rhead{\thepage}
  \renewcommand{\headrulewidth}{0pt}
}

% === Standard Spacing for Tight Layout ===
% Use safe positive values to prevent the layout engine from breaking.
\setlength{\headheight}{14pt}  % Space for the page number itself
\setlength{\headsep}{10pt}     % Gap between page number and the first line of content
\setlength{\footskip}{0pt}     % No footer space needed since we use no footers
